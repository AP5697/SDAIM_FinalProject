# Development-only dependencies: training, experiment tracking and testing.
#
# Deliberately separate from requirements.txt, which is installed by the Hugging
# Face Space. Nothing here reaches the deployed application - MLflow records how
# the model was arrived at, which is a build-time concern, and shipping it would
# add a large dependency the Space never imports.
#
# Install with:
#     pip install -r requirements.txt -r requirements-dev.txt

# Figure generation in mlops/model_building/visualisation.py, which train.py
# and the EDA script import at module level. Undeclared until now: they were
# present locally, and matplotlib happened to arrive transitively via mlflow,
# so the gap only appeared once CI ran training in a clean environment and the
# `import seaborn` failed. The deployed app imports neither - it never touches
# visualisation.py - which is why they belong here and not in requirements.txt.
matplotlib==3.11.0
seaborn==0.13.2

# Experiment tracking. mlops/model_building/tracking.py degrades to a no-op
# without it, so the training run still succeeds if this is missing.
mlflow==3.14.0

# NOTE: `shap` is deliberately not listed, in development either.
#
# Its compiled extension segfaults on Linux against this xgboost and numpy
# build. That took down the deployed Space once, and then the CI training job:
# a segfault kills the interpreter, so the try/except guarding the import never
# runs and the whole process dies. Both the per-session attribution in the app
# and the global importance table in training now use XGBoost's own exact
# TreeSHAP (`pred_contribs=True`), which produces identical values with no
# extra native library to disagree with anything.

# Test suite.
pytest==9.1.1
